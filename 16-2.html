JavaScript 对象演示

字面量创建 / 属性访问 / 方法与 this / 遍历 / 构造函数 / 对象数组统计 / 增删改

一、用对象字面量创建对象
对象是“键值对”的集合，花括号内用逗号分隔各个属性，属性值可以是数字、字符串，也可以是另一个对象。
var guan_student = { guan_name: "关斯译", guan_id: "24001010318", guan_class: "计算机24003班", guan_scores: { guan_web: 92, guan_js: 88 } // 嵌套对象 };
二、两种属性访问方式
点语法简单直观，适合固定的属性名；方括号语法可以拼接变量、读取动态属性名。
guan_student.guan_name // 点语法 guan_student["guan_id"] // 方括号语法 guan_student.guan_scores.guan_web // 逐层访问嵌套对象 var guan_key = "guan_class"; guan_student[guan_key] // 用变量取属性
三、对象的方法与 this
属性值是函数时称为“方法”；方法内部的 this 指向调用它的那个对象，因此可以直接读取同一对象上的属性。
var guan_report = { guan_owner: "关斯译", guan_scores: [92, 88, 95], guan_total: function () { var guan_sum = 0; for (var guan_i = 0; guan_i < this.guan_scores.length; guan_i++) { guan_sum += this.guan_scores[guan_i]; } return guan_sum; }, guan_average: function () { return (this.guan_total() / this.guan_scores.length).toFixed(1); } };
四、遍历对象：for...in 与 Object.keys
for...in 逐个取出属性名（键），再用方括号取出对应的值；Object.keys / Object.values 直接把键、值收集成数组。
for (var guan_key in guan_student) { ... } Object.keys(guan_student); // ["guan_name", "guan_id", ...] Object.values(guan_student); // ["关斯译", "24001010318", ...]
五、构造函数批量创建对象
把“创建对象”的过程写成构造函数，配合 new 就能按同一模板生成多个结构一致的对象实例。
function guan_Student(guan_name, guan_id, guan_scores) { this.guan_name = guan_name; this.guan_id = guan_id; this.guan_scores = guan_scores; this.guan_average = function () { var guan_sum = 0; for (var guan_i = 0; guan_i < this.guan_scores.length; guan_i++) { guan_sum += this.guan_scores[guan_i]; } return guan_sum / this.guan_scores.length; }; } var guan_stu1 = new guan_Student("关斯译", "24001010318", [92, 88, 95]);
六、对象数组的统计与排序
对象数组是最常用的数据结构：先用 sort 按平均分从高到低排序，再汇总出最高分与班级平均分。
七、为对象增、删、改属性
对象是动态的：随时可以添加新属性、修改已有属性，或用 delete 删除属性。下面的按钮会直接操作 guan_student 对象。
guan_student.guan_email = "guan@example.com"; // 增加 guan_student.guan_class = "计算机24003班"; // 修改 delete guan_student.guan_email; // 删除
